 30def _check_url(value: str) -> str:
 31    value = value.strip()
 32    if not re.match(r"^https?://[^\s/$.?#].[^\s]*$", value, re.IGNORECASE):

 37def _clean_str_list(values: list[str]) -> list[str]:
 38    """Strip, drop blanks, and de-duplicate (case-insensitive) while keeping order."""
 39    seen: set[str] = set()
 40    cleaned = []
 41    for value in values:
 42        value = value.strip()
 43        if value and value.lower() not in seen:
 44            seen.add(value.lower())
 45            cleaned.append(value)
 46    return cleaned

 88    @field_validator("password")
 89    @classmethod
 90    def password_strength(cls, value: str) -> str:
 91        # R1.4 minimum security requirements. bcrypt only reads the first 72 bytes,
 92        # so longer passwords are rejected rather than silently truncated.
 93        if len(value) < 8:
 94            raise ValueError("Password must be at least 8 characters long")
 95        if len(value.encode("utf-8")) > 72:
 96            raise ValueError("Password must be at most 72 bytes long")
 97        if not re.search(r"[A-Za-z]", value) or not re.search(r"\d", value):
 98            raise ValueError("Password must contain at least one letter and one number")
 99        return value
