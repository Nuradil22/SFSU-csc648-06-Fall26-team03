  55def clean_resources(raw) -> list[dict]:
  56    """Learning resources: [{title, url, provider?, type?, description?}].
  57
  58    Only https links with a host and a title are kept, so a roadmap can never
  59    carry javascript:, data:, or plain-http links. Unknown fields are dropped.
  60    """
  61    resources = []
  62
  63    for item in raw if isinstance(raw, list) else []:
  64        if not isinstance(item, dict):
  65            continue
  66
  67        title = str(item.get("title") or "").strip()[:200]
  68        url = str(item.get("url") or "").strip()
  69
  70        try:
  71            parts = urlsplit(url)
  72        except ValueError:
  73            continue
  74
  75        if not title or len(url) > 2000 or parts.scheme != "https" or not parts.hostname:
  76            continue
  77
  78        resource = {"title": title, "url": url}
  79
  80        for key, limit in (("provider", 100), ("type", 100), ("description", 300)):
  81            value = str(item.get(key) or "").strip()[:limit]
  82            if value:
  83                resource[key] = value
  84
  85        resources.append(resource)
  86
  87        if len(resources) == MAX_RESOURCES_PER_MILESTONE:
  88            break
  89
  90    return resources

 182def search_sources(response, domains, blocked=None) -> tuple[int, dict]:
 183    """(number of web searches run, {url key: url}) for one Responses result.
 184
 185    Sources come from the web_search_call items (requested with
 186    include=["web_search_call.action.sources"]) and from any url_citation
 187    annotations; with an allowlist, only those on an allowed domain count,
 188    and sources on a blocked domain never count.
 189    """
 190    blocked = blocked_domains() if blocked is None else blocked
 191    searches = 0
 192    found = []
 193
 194    for item in _field(response, "output") or []:
 195        kind = _field(item, "type")
 196
 197        if kind == "web_search_call":
 198            searches += 1
 199            action = _field(item, "action")
 200            found.append(_field(action, "url"))  # "open_page" actions
 201            found.extend(_field(source, "url") for source in _field(action, "sources") or [])
 202
 203        elif kind == "message":
 204            for part in _field(item, "content") or []:
 205                for note in _field(part, "annotations") or []:
 206                    if _field(note, "type") == "url_citation":
 207                        found.append(_field(note, "url"))
 208
 209    sources = {}
 210    for url in found:
 211        key = _url_key(url)
 212        if (key and _host_allowed(key, domains)
 213                and not any(_on_domain(key[0], domain) for domain in blocked)):
 214            sources.setdefault(key, str(url).strip())
 215
 216    return searches, sources

 219# --- Link check -----------------------------------------------------------------
 220#
 221# A link the search returned is only shown if it also opens over valid HTTPS
 222# from the server. The check is a plain GET (headers only; no page content is
 223# read or kept) with normal certificate verification and no attempt to get
 224# around a site's access rules. Results:
 225#   ok           2xx                                    -> shown
 226#   blocked      401/403/429/451: the site refuses      -> shown only because
 227#                automated checks; proves nothing          the search returned it
 228#   missing      404/410                                -> dropped
 229#   unavailable  TLS/connection error, timeout, 5xx,    -> dropped
 230#                broken/insecure/too many redirects,
 231#                or a non-public address
 232
 233LINK_OK, LINK_BLOCKED, LINK_MISSING, LINK_UNAVAILABLE = "ok", "blocked", "missing", "unavailable"
 234LINK_DROP = {LINK_MISSING, LINK_UNAVAILABLE}
 235_MAX_REDIRECTS = 3
 236_MAX_LINKS_CHECKED = 20
 237# Tests replace this with an httpx.MockTransport; None means real network.
 238_link_transport = None

 241def _public_host(host: str) -> bool:
 242    """Only public internet addresses are contacted (no internal services)."""
 243    try:
 244        infos = socket.getaddrinfo(host, 443, proto=socket.IPPROTO_TCP)
 245    except (OSError, UnicodeError):
 246        return False
 247    return bool(infos) and all(ipaddress.ip_address(info[4][0]).is_global for info in infos)

 250def check_link(client, url: str) -> str:
 251    for _ in range(_MAX_REDIRECTS + 1):
 252        parts = urlsplit(url)
 253        if parts.scheme != "https" or not parts.hostname or not _public_host(parts.hostname):
 254            return LINK_UNAVAILABLE
 255
 256        try:
 257            with client.stream("GET", url) as response:
 258                status = response.status_code
 259                location = response.headers.get("location")
 260        except httpx.HTTPError:
 261            return LINK_UNAVAILABLE
 262
 263        if 300 <= status < 400 and location:
 264            url = urljoin(url, location)
 265            continue
 266        if 200 <= status < 300:
 267            return LINK_OK
 268        if status in (404, 410):
 269            return LINK_MISSING
 270        if status in (401, 403, 429, 451):
 271            return LINK_BLOCKED
 272        return LINK_UNAVAILABLE
 273
 274    return LINK_UNAVAILABLE  # too many redirects

 277def check_links(urls) -> dict:
 278    """{url: result} for up to _MAX_LINKS_CHECKED links, checked in parallel.
 279    Never raises; a link that can't be checked is "unavailable"."""
 280    urls = list(dict.fromkeys(urls))[:_MAX_LINKS_CHECKED]
 281    if not urls:
 282        return {}
 283
 284    def one(url):
 285        try:
 286            return check_link(client, url)
 287        except Exception:  # noqa: BLE001 - a failed check must not fail the roadmap
 288            return LINK_UNAVAILABLE
 289
 290    with httpx.Client(
 291        transport=_link_transport,
 292        timeout=settings.resource_link_check_timeout,
 293        follow_redirects=False,
 294        headers={"User-Agent": "Projectory-LinkCheck/1.0"},
 295    ) as client, ThreadPoolExecutor(max_workers=8) as pool:
 296        return dict(zip(urls, pool.map(one, urls)))

 299def drop_unreachable(sources: dict, candidate_urls) -> dict:
 300    """Remove sources the model recommended whose link doesn't open."""
 301    keys = {_url_key(url) for url in candidate_urls} & sources.keys()
 302    if not settings.resource_link_check or not keys:
 303        return sources
 304
 305    results = check_links(sources[key] for key in keys)
 306    dropped = {key for key in keys if results.get(sources[key], LINK_UNAVAILABLE) in LINK_DROP}
 307    logger.info(
 308        "Resource link check: checked=%d dropped=%d blocked=%d",
 309        len(results), len(dropped), sum(r == LINK_BLOCKED for r in results.values())
 310    )
 311    return {key: url for key, url in sources.items() if key not in dropped}

 332def verified_resources(raw, allowed: dict) -> list[dict]:
 333    """Keep only resources whose URL is in `allowed`, using the source's own
 334    URL. Duplicates and anything unsupported are dropped. `type` marks
 335    official SFSU pages ("sfsu") versus other providers ("external")."""
 336    kept, seen = [], set()
 337
 338    for item in raw or []:
 339        item = item.model_dump() if isinstance(item, BaseModel) else item
 340        if not isinstance(item, dict):
 341            continue
 342
 343        key = _url_key(item.get("url"))
 344        if key not in allowed or key in seen:
 345            continue
 346        seen.add(key)
 347
 348        kept.append({
 349            "title": item.get("name") or item.get("title"),
 350            "description": item.get("description"),
 351            "provider": item.get("provider"),
 352            "url": allowed[key],
 353            "type": "sfsu" if _on_domain(key[0], SFSU_DOMAIN) else "external",
 354        })
 355
 356    return clean_resources(kept)

 359def remove_unverified_urls(text: str, allowed: dict) -> str:
 360    def replace(match):
 361        raw = match.group(0)
 362        url = raw.rstrip(".,;:!?")
 363        if _url_key(url) in allowed:
 364            return raw
 365        return "[unverified link removed]" + raw[len(url):]
 366
 367    return _URL_IN_TEXT.sub(replace, text)

 389    goals = []
 390
 391    for goal in body.goals:
 392        milestones = []
 393
 394        for raw in goal.milestones:
 395            title = str(raw.get("title", "")).strip()[:200]
 396
 397            if not title:

 399
 400            milestone = {
 401                "id": str(raw.get("id") or uuid4())[:80],
 402                "title": title,
 403                "completed": bool(raw.get("completed", False))
 404            }
 405
 406            # Kept through owner edits (PUT /career/roadmap) so ticking a
 407            # milestone never erases its links.
 408            resources = clean_resources(raw.get("resources"))
 409            if resources:
 410                milestone["resources"] = resources
 411
 412            milestones.append(milestone)
 413
 414        goals.append({
 415            "id": goal.id,
 416            "title": goal.title,
 417            "milestones": milestones
 418        })
 419
 420    return {"title": body.title, "goals": goals}

 721def _response_text(response) -> tuple[str, str | None]:
 722    """(output text, refusal or None) from a Responses API result."""
 723    texts, refusal = [], None
 724
 725    for item in _field(response, "output") or []:
 726        if _field(item, "type") != "message":
 727            continue
 728        for part in _field(item, "content") or []:
 729            kind = _field(part, "type")
 730            if kind == "output_text":
 731                texts.append(_field(part, "text") or "")
 732            elif kind == "refusal":
 733                refusal = _field(part, "refusal") or "refused"
 734
 735    return "".join(texts), refusal

 750def generate_openai_answer(message, payload, history, roadmap, pending=None, readiness=None,
 751                           mode=None, portfolio_changed=False):
 752    import openai
 753    from openai.lib._parsing._responses import type_to_text_format_param
 754
 755    if not settings.openai_api_key:
 756        raise HTTPException(
 757            status_code=503,
 758            detail="OpenAI is not configured on the server."
 759        )
 760
 761    domains = research_domains()
 762    # Roadmap generation always researches (when enabled); ordinary chat only
 763    # for questions about courses, programs, certificates, or resources.
 764    research = settings.openai_web_search and (mode is not None or wants_research(message))
 765
 766    client = openai.OpenAI(
 767        api_key=settings.openai_api_key,
 768        timeout=settings.ai_timeout_seconds,
 769        # A retried research call can repeat billed searches.
 770        max_retries=1 if research else 2
 771    )
 772
 773    system = (
 774        build_system_prompt(payload, roadmap, pending, readiness)
 775        + roadmap_instructions(mode, portfolio_changed)
 776        + research_instructions(research, domains)
 777    )
 778
 779    messages = [
 780        {"role": item.role, "content": item.content}
 781        for item in history[-12:]
 782    ]
 783
 784    messages.append({
 785        "role": "user",
 786        "content": message
 787    })
 788
 789    options = {}
 790    if research:
 791        tool = {
 792            "type": "web_search",
 793            "search_context_size": settings.openai_web_search_context,
 794        }
 795        if domains:
 796            tool["filters"] = {"allowed_domains": domains}
 797        options = {
 798            "tools": [tool],
 799            "include": ["web_search_call.action.sources"],
 800            "max_tool_calls": settings.openai_max_web_searches,
 801        }
 802
 803    max_output_tokens = (
 804        settings.openai_max_output_tokens_roadmap if mode
 805        else settings.openai_max_output_tokens_chat
 806    )
 807
 808    # responses.create (not .parse) with the same strict JSON schema: the
 809    # response object is always returned, so status, usage, and searches are
 810    # known (and logged) even when the output is cut off or unusable.
 811    try:
 812        response = client.responses.create(
 813            model=settings.openai_model,
 814            instructions=system,
 815            input=messages,
 816            text={"format": type_to_text_format_param(AIChatResponse)},
 817            reasoning={"effort": settings.openai_reasoning_effort},
 818            max_output_tokens=max_output_tokens,
 819            **options
 820        )
 821
 822    except openai.AuthenticationError as exc:
 823        logger.error("OpenAI authentication failed: %s", type(exc).__name__)
 824        raise AIResponseError(503, "OpenAI authentication failed on the server.") from exc
 825
 826    except openai.RateLimitError as exc:
 827        logger.warning("OpenAI rate limit reached.")
 828        raise AIResponseError(503, "OpenAI is temporarily rate-limited or out of credits.") from exc
 829
 830    except openai.APIError as exc:
 831        # Log only OpenAI's error metadata; never the request or user data.
 832        logger.error(
 833            "OpenAI API failed: %s status=%s code=%s param=%s",
 834            type(exc).__name__,
 835            getattr(exc, "status_code", None),
 836            getattr(exc, "code", None),
 837            getattr(exc, "param", None)
 838        )
 839        raise AIResponseError(503, "The AI service is unavailable. Please retry.") from exc
 840
 841    searches, sources = search_sources(response, domains) if research else (0, {})
 842    status = _field(response, "status") or "completed"
 843    usage = _field(response, "usage")
 844    logger.info(
 845        "OpenAI chat: status=%s web_searches=%d sources=%d input_tokens=%s output_tokens=%s",
 846        status, searches, len(sources),
 847        _field(usage, "input_tokens"), _field(usage, "output_tokens")
 848    )
 849
 850    if status == "incomplete":
 851        reason = _field(_field(response, "incomplete_details"), "reason")
 852        logger.warning(
 853            "OpenAI response incomplete: reason=%s output_tokens=%s limit=%d",
 854            reason, _field(usage, "output_tokens"), max_output_tokens
 855        )
 856        raise AIResponseError(
 857            502, _incomplete_detail(reason, mode),
 858            _usage_counts(response, searches, "incomplete")
 859        )
 860
 861    if status != "completed":
 862        logger.warning("OpenAI response status: %s", status)
 863        raise AIResponseError(
 864            502, "The AI response could not be completed. Please retry.",
 865            _usage_counts(response, searches, str(status)[:20])
 866        )
 867
 868    text, refusal = _response_text(response)
 869
 870    if refusal:
 871        raise AIResponseError(
 872            502, "The AI declined to answer this request. Please rephrase it.",
 873            _usage_counts(response, searches, "invalid")
 874        )
 875
 876    try:
 877        result = AIChatResponse.model_validate_json(text)
 878
 879        # Recommended links must also open (checked server-side, see above).
 880        recommended = [item.url for item in result.resources]
 881        if result.proposal is not None:
 882            recommended += [
 883                item.url
 884                for goal in result.proposal.goals
 885                for milestone in goal.milestones
 886                for item in milestone.resources
 887            ]
 888        sources = drop_unreachable(sources, recommended)
 889
 890        # Links the student already has may be kept in a revision.
 891        allowed = {**known_resource_urls(roadmap, pending), **sources}
 892
 893        proposal = None
 894
 895        if result.proposal is not None:
 896            # Enforce the existing RoadmapBody contract (lengths, counts).
 897            # pydantic.ValidationError is a ValueError -> 502 below.
 898            proposal = clean_plan(
 899                RoadmapBody.model_validate(
 900                    verify_plan_resources(result.proposal.model_dump(), allowed)
 901                )
 902            )
 903
 904        return _answer(
 905            remove_unverified_urls(result.reply, allowed)[:5000],
 906            proposal,
 907            # New recommendations must come from this request's search.
 908            verified_resources(result.resources, sources),
 909            searches,
 910            usage,
 911        ), "openai"
 912
 913    except (ValueError, TypeError) as exc:
 914        logger.warning("Invalid OpenAI response: %s", type(exc).__name__)
 915        raise AIResponseError(
 916            502, "The AI response could not be processed. Please retry.",
 917            _usage_counts(response, searches, "invalid")
 918        ) from exc
